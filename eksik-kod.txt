        function startCountdown(){
            const target=new Date("2036-08-01T00:00:00");
            function tick(){
                const diff=target-new Date(); if(diff<=0) return;
                document.getElementById("cd-gun").textContent=String(Math.floor(diff/86400000)).padStart(2,"0");
                document.getElementById("cd-saat").textContent=String(Math.floor((diff%86400000)/3600000)).padStart(2,"0");

                document.getElementById("cd-san").textContent=String(Math.floor((diff%60000)/1000)).padStart(2,"0");
            }
            tick(); setInterval(tick,1000);
        }

const bikeColors={Ducati:"#cc0000",Yamaha:"#0055cc",KTM:"#e55a00",Honda:"#e8001d",Aprilia:"#1a1a6e",BMW:"#0066cc",Kawasaki:"#00aa00",Bimota:"#006600",Moto2:"#0099ff",Moto3:"#00cc66",SSP:"#ff6600",WCR:"#cc44aa",F1:"#e10600",default:"#334466"};

function buildRiderSection(secId){
    const sec=riderSections[secId]; if(!sec) return;
    const el=document.getElementById(secId);
    const filterBikes=[...new Set(sec.riders.map(r=>r.bike))];
    let filterHTML='<button class="rss-tab active" onclick="filterRidersInSection(\''+secId+'\',\'ALL\',this)">Tümü</button>';
    filterBikes.forEach(bk=>{
        const bc=bikeColors[bk]||bikeColors.default;
        filterHTML+=`<button class="rss-tab" onclick="filterRidersInSection('${secId}','${bk}',this)" style="--fc:${bc}">${bk}</button>`;
    });
    el.innerHTML=`
        <div class="rider-section-header">
            <div>
                <div class="rider-section-title" style="color:${sec.color}">${sec.title}</div>
                <div class="rider-section-sub">${sec.subtitle}</div>
            </div>
        </div>
        <div class="rider-filter-bar" id="rfbar-${secId}">${filterHTML}</div>
        <div class="rider-grid" id="rgrid-${secId}"></div>
        <p style="margin-top:12px;font-size:.68rem;color:var(--muted);text-align:right;">MOTOGPVIDEO.COM · 2026</p>`;
    renderRiders(secId,sec.riders);
}

function renderRiders(secId,list){
    const grid=document.getElementById("rgrid-"+secId); if(!grid) return;
    grid.innerHTML=list.map(r=>{
        const bc=bikeColors[r.bike]||bikeColors.default;
        const isFeatured=r.featured||false;
        const topGrad=isFeatured
            ?`background:linear-gradient(135deg,#1a0040 0%,#4a0066 60%,#cc0080 100%);`
            :`background:linear-gradient(135deg,#0d1220 0%,${bc}44 100%);`;
        return `<div class="rider-card${isFeatured?" featured":""}">
            <div class="rider-card-top" style="${topGrad}">
                <span class="rider-num-bg">${r.num}</span>
                <span class="rider-num-badge" style="color:${isFeatured?"#ff44cc":"#fff"}">${r.num}</span>
                <span class="rider-flag-badge">${r.flag}</span>
            </div>
            <div class="rider-card-body">
                <div class="rider-name">${r.name}</div>
                <div class="rider-country">${r.country}</div>
                <div class="rider-team" style="border-color:${bc}">${r.team}</div>
                <span class="rider-bike-tag" style="background:${bc}99;border:1px solid ${bc}">${r.bike}</span>
            </div>
        </div>`;
    }).join("");
}

function filterRidersInSection(secId,bike,btn){
    document.querySelectorAll("#rfbar-"+secId+" .rss-tab").forEach(b=>b.classList.remove("active"));
    btn.classList.add("active");
    const sec=riderSections[secId]; if(!sec) return;
    const filtered=bike==="ALL"?sec.riders:sec.riders.filter(r=>r.bike===bike);
    renderRiders(secId,filtered);
}

function fmtDate(str){if(!str)return"";const d=new Date(str);if(isNaN(d))return str;return d.toLocaleDateString("tr-TR",{day:"numeric",month:"long",year:"numeric"});}
function stripHtml(html){const t=document.createElement("div");t.innerHTML=html||"";return t.textContent||"";}
async function loadRSS(btn){
    document.querySelectorAll(".rss-tab[data-feed]").forEach(t=>t.classList.remove("active"));
    btn.classList.add("active");
    const url=btn.dataset.feed;
    const statusEl=document.getElementById("rss-status");
    const gridEl=document.getElementById("news-grid");
    statusEl.innerHTML='<div class="rss-spinner"></div> Haberler yükleniyor…';
    gridEl.innerHTML=Array(6).fill('<div class="news-card skeleton"></div>').join("");
    try{
        const res=await fetch(url); const data=await res.json();
        if(!data.items||!data.items.length) throw new Error();
        const items=data.items.slice(0,12);
        statusEl.innerHTML=`<span style="color:#8aaccc">${items.length} haber yüklendi</span>`;
        gridEl.innerHTML=items.map(item=>{
            const thumb=item.thumbnail||item.enclosure?.link||"";
            const cat=item.categories?.[0]||"Haber";
            const desc=stripHtml(item.description).slice(0,80).trim();
            return`<div class="news-card">
                ${thumb?`<img class="news-thumb" src="${thumb}" alt="" loading="lazy" onerror="this.remove()">`:""}
                <div class="news-cat">${cat}</div>
                <div class="news-title">${item.title}</div>
                ${desc?`<div style="font-size:.72rem;color:var(--muted);line-height:1.5">${desc}…</div>`:""}
                <div class="news-footer">
                    <span class="news-date">${fmtDate(item.pubDate)}</span>
                    <a href="${item.link}" target="_blank" class="btn" style="padding:5px 10px;font-size:.62rem;width:auto">OKU</a>
                </div>
            </div>`;
        }).join("");
    }catch{
        statusEl.style.color="#ff6680";
        statusEl.innerHTML="⚠ RSS yüklenemedi";
        gridEl.innerHTML=`
            <div class="card" style="align-items:flex-start;gap:8px"><h3>MotoGP Haberleri</h3><a href="https://www.motogp.com/en/news" target="_blank" class="btn">Siteye Git</a></div>

    }
}
